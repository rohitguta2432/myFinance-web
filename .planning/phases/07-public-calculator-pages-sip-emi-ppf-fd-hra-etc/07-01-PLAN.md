**Financial formulas (all return number):**
```typescript
// SIP future value: M * [((1 + r)^n - 1) / r] * (1 + r)
// where r = annualRate/12/100, n = years*12, M = monthlyAmount
export function calcSIP(monthlyAmount: number, annualRate: number, years: number): number

// Lumpsum future value: P * (1 + r)^n
// where r = annualRate/100, n = years
export function calcLumpsum(principal: number, annualRate: number, years: number): number

// EMI: P * r * (1+r)^n / ((1+r)^n - 1)
// where r = annualRate/12/100, n = tenureMonths
export function calcEMI(principal: number, annualRate: number, tenureMonths: number): number

// FD maturity: P * (1 + r/freq)^(freq*years)
// where freq = compounding frequency (4 = quarterly, 2 = half-yearly, 1 = annual)
export function calcFD(principal: number, annualRate: number, years: number, freq?: number): number

// PPF maturity after 15 years: year-by-year accumulation with 7.1% annual compounding
// yearly deposit is made at start of each year
export function calcPPF(yearlyDeposit: number, years?: number): number

// HRA exemption: min of (actualHRA, rent-10%salary, 50%salary for metro / 40% non-metro)
export function calcHRA(basicSalary: number, hra: number, rentPaid: number, isMetro: boolean): number

// NPS corpus at retirement: SIP formula with annualRate=10%, then annuity at 6% for remaining years
// returns { corpus, monthlyPension }
export function calcNPS(monthlyContribution: number, currentAge: number, retirementAge?: number): { corpus: number; monthlyPension: number }

// Retirement required corpus: (monthlyExpenses * 12) / withdrawalRate
// where withdrawalRate defaults to 0.04 (4% safe withdrawal)
// then adjust for inflation: requiredCorpus / (1+inflationRate)^yearsToRetire
export function calcRetirement(monthlyExpenses: number, currentAge: number, retirementAge: number, inflationRate?: number): number

// SWP sustainability: how long corpus lasts at monthly withdrawal with r% growth
// Returns years the corpus lasts (capped at 50 years)
export function calcSWP(corpus: number, monthlyWithdrawal: number, annualGrowthRate: number): number

// Inflation future cost: cost * (1 + rate/100)^years
export function calcInflation(currentCost: number, inflationRate: number, years: number): number
```
