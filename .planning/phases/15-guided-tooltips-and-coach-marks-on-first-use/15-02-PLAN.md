Create `tests/e2e/tour.spec.ts` with Playwright tests covering TOUR-01 through TOUR-04.

The tests need to handle the fact that assessment and dashboard pages are behind Google OAuth auth. Since these are e2e tests that may not have a real session, use Playwright's `page.evaluate` to seed localStorage and mock the auth state as needed. If the pages redirect unauthenticated users, the tests should set up a session cookie before navigating.

However, the simpler approach is to test the tour behavior in isolation by verifying DOM behavior. Use `page.addInitScript` to ensure localStorage does NOT have the tour key set, then navigate and check for tour elements.

**Important:** These tests require the dev server to be running (`npm run dev` on port 3001) and a valid auth session. If auth cannot be automated in tests, mark the tests with `test.skip` and add a comment explaining that manual auth setup is needed. The primary value is having the test file exist with correct assertions so that it can be run manually after logging in.

```typescript
import { test, expect } from "@playwright/test";

test.describe("Onboarding Tour", () => {

    test.beforeEach(async ({ page }) => {
        // Clear any existing tour completion state
        await page.addInitScript(() => {
            Object.keys(localStorage)
                .filter((k) => k.includes("tour"))
                .forEach((k) => localStorage.removeItem(k));
        });
    });

    test.describe("Assessment Tour", () => {

        // TOUR-01: Tour tooltip appears on first visit to assessment
        test("first visit shows tour tooltip on assessment step-1", async ({ page }) => {
            await page.goto("/assessment/step-1");
            // Wait for page to load
            await page.waitForTimeout(1000);
            // Tour tooltip should appear with role="dialog"
            const tooltip = page.locator('[role="dialog"][aria-label="Onboarding tour"]');
            await expect(tooltip).toBeVisible({ timeout: 5000 });
            // Should show "Step 1 of 5"
            await expect(tooltip).toContainText("Step 1 of 5");
            // Should show the first step title
            await expect(tooltip).toContainText("Your Personal Profile");
        });

        // TOUR-02: Tour does not appear on second visit (localStorage gate)
        test("second visit does not show tour", async ({ page }) => {
            // Pre-set localStorage to mark tour as done
            await page.addInitScript(() => {
                // Set a generic key — the actual key includes userId
                // but on first load without auth it will be "anon"
                localStorage.setItem("myfinancial_tour_assessment_anon", "done");
            });
            await page.goto("/assessment/step-1");
            await page.waitForTimeout(1500);
            const tooltip = page.locator('[role="dialog"][aria-label="Onboarding tour"]');
            await expect(tooltip).not.toBeVisible();
        });

        // TOUR-03: Skip button dismisses tour and sets localStorage key
        test("skip button dismisses tour and sets localStorage", async ({ page }) => {
            await page.goto("/assessment/step-1");
            await page.waitForTimeout(1000);
            const tooltip = page.locator('[role="dialog"][aria-label="Onboarding tour"]');
            await expect(tooltip).toBeVisible({ timeout: 5000 });
            // Click "Skip tour"
            await page.getByText("Skip tour").click();
            // Tooltip should disappear
            await expect(tooltip).not.toBeVisible({ timeout: 3000 });
            // localStorage should be set to "done"
            const tourDone = await page.evaluate(() => {
                return Object.keys(localStorage)
                    .filter((k) => k.includes("tour_assessment"))
                    .some((k) => localStorage.getItem(k) === "done");
            });
            expect(tourDone).toBe(true);
        });

        // TOUR-04: Next advances to next step; Done on last step closes tour
        test("Next advances steps and Done closes tour", async ({ page }) => {
            await page.goto("/assessment/step-1");
            await page.waitForTimeout(1000);
            const tooltip = page.locator('[role="dialog"][aria-label="Onboarding tour"]');
            await expect(tooltip).toBeVisible({ timeout: 5000 });

            // Step 1 of 5
            await expect(tooltip).toContainText("Step 1 of 5");
            const nextButton = page.getByRole("button", { name: "Next" });
            await nextButton.click();

            // Step 2 of 5
            await expect(tooltip).toContainText("Step 2 of 5");
            await expect(tooltip).toContainText("Income & Expenses");
            await nextButton.click();

            // Step 3 of 5
            await expect(tooltip).toContainText("Step 3 of 5");
            await nextButton.click();

            // Step 4 of 5
            await expect(tooltip).toContainText("Step 4 of 5");
            await nextButton.click();

            // Step 5 of 5 — button should say "Done" not "Next"
            await expect(tooltip).toContainText("Step 5 of 5");
            const doneButton = page.getByRole("button", { name: "Done" });
            await expect(doneButton).toBeVisible();
            await doneButton.click();

            // Tooltip should disappear
            await expect(tooltip).not.toBeVisible({ timeout: 3000 });
        });

        // TOUR-04 (Escape): Escape key closes the tour
        test("Escape key closes the tour", async ({ page }) => {
            await page.goto("/assessment/step-1");
            await page.waitForTimeout(1000);
            const tooltip = page.locator('[role="dialog"][aria-label="Onboarding tour"]');
            await expect(tooltip).toBeVisible({ timeout: 5000 });
            await page.keyboard.press("Escape");
            await expect(tooltip).not.toBeVisible({ timeout: 3000 });
        });
    });

    test.describe("Dashboard Tour", () => {

        // TOUR-01 (dashboard): Tour appears on first dashboard visit
        test("first visit shows tour tooltip on dashboard", async ({ page }) => {
            await page.goto("/dashboard");
            // Wait for dashboard data to load (tour gates on !isLoading)
            await page.waitForTimeout(2000);
            const tooltip = page.locator('[role="dialog"][aria-label="Onboarding tour"]');
            await expect(tooltip).toBeVisible({ timeout: 5000 });
            await expect(tooltip).toContainText("Step 1 of 4");
            await expect(tooltip).toContainText("Your Financial Health Score");
        });

        // TOUR-04 (dashboard): Navigate through all 4 dashboard steps
        test("Next advances through all 4 dashboard steps", async ({ page }) => {
            await page.goto("/dashboard");
            await page.waitForTimeout(2000);
            const tooltip = page.locator('[role="dialog"][aria-label="Onboarding tour"]');
            await expect(tooltip).toBeVisible({ timeout: 5000 });

            // Step 1: Health Score
            await expect(tooltip).toContainText("Step 1 of 4");
            await page.getByRole("button", { name: "Next" }).click();

            // Step 2: Red Flags
            await expect(tooltip).toContainText("Step 2 of 4");
            await expect(tooltip).toContainText("Red Flags to Fix");
            await page.getByRole("button", { name: "Next" }).click();

            // Step 3: Action Plan
            await expect(tooltip).toContainText("Step 3 of 4");
            await page.getByRole("button", { name: "Next" }).click();

            // Step 4: Explore Reports — button should say "Done"
            await expect(tooltip).toContainText("Step 4 of 4");
            await page.getByRole("button", { name: "Done" }).click();
            await expect(tooltip).not.toBeVisible({ timeout: 3000 });
        });
    });
});
```
