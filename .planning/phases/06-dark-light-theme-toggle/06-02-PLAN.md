```typescript
export interface AppPalette {
    bg: string;     // Page background
    s1: string;     // Surface 1 (card, panel)
    s2: string;     // Surface 2 (inner card, input bg)
    s3: string;     // Surface 3 (hover bg, active chip)
    txt: string;    // Primary text
    txt2: string;   // Secondary text
    mute: string;   // Muted / tertiary text
    accent: string; // Emerald #10B981 (same in both themes)
    brd: string;    // Border subtle
    brd2: string;   // Border emphasis
    danger: string; // Error / destructive
    warn: string;   // Warning
}

export const DARK_PALETTE: AppPalette;   // dark hex values
export const LIGHT_PALETTE: AppPalette;  // light hex values

export function useAppTheme(): AppPalette;
// Usage: const palette = useAppTheme();
// Returns DARK_PALETTE until hydration, then the resolved theme's palette.
```
